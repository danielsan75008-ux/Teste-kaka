local function getParentGui()
    -- 1. gethui / get_hidden_gui
    local ok, hui = pcall(function()
        local f = rawget(getfenv(), "gethui") or (getgenv and rawget(getgenv(), "gethui"))
        if typeof(f) == "function" then return f() end
        local f2 = rawget(getfenv(), "get_hidden_gui") or (getgenv and rawget(getgenv(), "get_hidden_gui"))
        if typeof(f2) == "function" then return f2() end
        return nil
    end)
    if ok and hui and typeof(hui) == "Instance" then
        pcall(function() 
            if syn and syn.protect_gui then syn.protect_gui(hui) end
        end)
        return hui
    end
    -- 2. CoreGui com cloneref
    local ok2, cg = pcall(function() return cloneref(CoreGui) end)
    if ok2 and cg then
        local testOk = pcall(function()
            local sg = Instance.new("ScreenGui")
            sg.Name = "AIChat_Test"
            sg.Parent = cg
            sg:Destroy()
        end)
        if testOk then return cg end
    end
    -- 3. PlayerGui
    if LocalPlayer then
        local ok3, pg = pcall(function() return LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:WaitForChild("PlayerGui", 2) end)
        if ok3 and pg then return pg end
    end
    -- 4. CoreGui direto
    if CoreGui then return CoreGui end
    return nil
end

-- Detecta funções executoras sem chamar nil
local function detectRequest()
    local names = {"request", "http_request"}
    for _, n in ipairs(names) do
        local ok, v = pcall(function() return rawget(getfenv(), n) end)
        if ok and typeof(v) == "function" then return v end
        local ok2, v2 = pcall(function() return getgenv and rawget(getgenv(), n) end)
        if ok2 and typeof(v2) == "function" then return v2 end
    end
    local ok, v = pcall(function() return syn and syn.request end)
    if ok and typeof(v) == "function" then return v end
    local ok2, v2 = pcall(function() return http and http.request end)
    if ok2 and typeof(v2) == "function" then return v2 end
    local ok3, v3 = pcall(function() return fluxus and fluxus.request end)
    if ok3 and typeof(v3) == "function" then return v3 end
    return nil
end

local function detectClipboard()
    local names = {"setclipboard", "toclipboard", "set_clipboard"}

local function hasFileFuncs()
    local ok1 = pcall(function() return typeof(isfolder) == "function" end)
    if not ok1 then return false end
    local a,b,c,d = false,false,false,false
    pcall(function() a = typeof(isfolder) == "function" end)
    pcall(function() b = typeof(isfile) == "function" end)
    pcall(function() c = typeof(readfile) == "function" end)
    pcall(function() d = typeof(writefile) == "function" end)
    return a and b and c and d
end

-- Storage
local Storage = {Folder="AIChat", File="AIChat/config.json"}
function Storage:load()
    if not hasFileFuncs() then return end
    local ok, exists = pcall(function() return isfile(self.File) end)
    if not ok or not exists then return end
    local ok2, content = pcall(function() return readfile(self.File) end)
    if not ok2 then return end
    local ok3, decoded = pcall(function() return HttpService:JSONDecode(content) end)
    if not ok3 or not decoded then return end
    if decoded.profiles then State.profiles = decoded.profiles end
    if decoded.activeProfileId then State.activeProfileId = decoded.activeProfileId end
    if decoded.systemPrompt ~= nil then State.systemPrompt = decoded.systemPrompt end
    if decoded.rscriptsApiKey ~= nil then State.rscriptsApiKey = decoded.rscriptsApiKey end
    if decoded.favorites then State.favorites = decoded.favorites end
    if decoded.customModels then State.customModels = decoded.customModels end
    if decoded.visual then
        if decoded.visual.accent then
            local c = decoded.visual.accent
            pcall(function() State.visual.accent = Color3.fromRGB(c.r,c.g,c.b) end)
        end
        if decoded.visual.transparency then State.visual.transparency = decoded.visual.transparency end
        if decoded.visual.scale then State.visual.scale = decoded.visual.scale end
        if decoded.visual.fontSize then State.visual.fontSize = decoded.visual.fontSize end
        if decoded.visual.rounding then State.visual.rounding = decoded.visual.rounding end
        if decoded.visual.animations ~= nil then State.visual.animations = decoded.visual.animations end
    end
end
function Storage:save()
    if not hasFileFuncs() then return false end
    pcall(function() if not isfolder(self.Folder) then makefolder(self.Folder) end end)
    local toSave = {
        profiles = State.profiles,

-- HTTP
local requestFunc = detectRequest()
local clipboardFunc = detectClipboard()

local function httpRequest(opts)
    if requestFunc then
        local ok, res = pcall(function() return requestFunc(opts) end)
        if ok and res then
            if typeof(res) == "string" then return {Success=true, StatusCode=200, Body=res} end
            return {Success=res.Success~=false, StatusCode=res.StatusCode or 200, Body=res.Body or "", Headers=res.Headers or {}}
        end
    end
    -- fallback RequestAsync (só GET funciona sem HttpEnabled em alguns jogos)
    local ok, res = pcall(function()
        return HttpService:RequestAsync({Url=opts.Url, Method=opts.Method or "GET", Headers=opts.Headers or {}, Body=opts.Body})
    end)
    if ok and res then return {Success=res.Success, StatusCode=res.StatusCode, Body=res.Body} end
    return {Success=false, StatusCode=0, Body="HTTP não disponível neste ambiente."}
end

-- RScripts Client (oficial)
local function rscriptsRequest(endpoint, params)
    if not State.rscriptsApiKey or State.rscriptsApiKey=="" then return false, "Configure sua RScripts API Key nas configurações." end
    local url = Config.RScriptsBase..endpoint
    if params and next(params) then
        local qs={}
        for k,v in pairs(params) do if v~=nil and v~="" then table.insert(qs, HttpService:UrlEncode(k).."="..HttpService:UrlEncode(tostring(v))) end end
        if #qs>0 then url=url.."?"..table.concat(qs,"&") end
    end
    local res = httpRequest({Url=url, Method="GET", Headers={["Authorization"]="Bearer "..State.rscriptsApiKey}})
    if not res.Success then
        local code=res.StatusCode or 0
        if code==401 then return false, "RScripts API Key inválida (401)." end
        if code==429 then return false, "Rate limit RScripts (429)." end
        return false, "Erro RScripts HTTP ("..code.."): "..tostring(res.Body):sub(1,200)
    end
    local ok, dec = pcall(function() return HttpService:JSONDecode(res.Body) end)
    if not ok then return false, "Resposta RScripts inválida (JSON)." end
    if dec.success==false then return false, "RScripts erro: "..tostring(dec.error and dec.error.message or "desconhecido") end
    return true, dec
end

    local sending=false
    local function onSend()
        if sending then return end
        local txt=inputBox.Text
        if not txt or txt:gsub("%s","")=="" then return end
        local prof=getActiveProfile()
        local ok, err = validateProfile(prof)
        if not ok then addMessage("system",err,true) return end
        addMessage("user",txt)
        table.insert(State.chatHistory,{role="user",content=txt})
        inputBox.Text=""
        sending=true sendBtn.Text="..." statusLbl.Text="Enviando..."
        task.spawn(function()
            local messages={}
            if State.systemPrompt~="" then table.insert(messages,{role="system",content=State.systemPrompt}) end
            for _,m in ipairs(State.chatHistory) do table.insert(messages,{role=m.role,content=m.content}) end
            local body=HttpService:JSONEncode({model=prof.MODEL, messages=messages})
            local res=httpRequest({Url=prof.API_URL, Method="POST", Headers={["Content-Type"]="application/json", ["Authorization"]="Bearer "..prof.API_KEY}, Body=body})
            sending=false sendBtn.Text="Enviar" statusLbl.Text=""
            if not res.Success then
                local code=res.StatusCode or 0
                local msg=res.Body or ""
                if code==401 then addMessage("system","API Key inválida (401).",true)
                elseif code==403 then addMessage("system","Acesso negado (403).",true)
                elseif code==404 then addMessage("system","Endpoint/modelo não encontrado (404).",true)
                elseif code==429 then addMessage("system","Rate limit (429).",true)
                else addMessage("system","Erro HTTP ("..code.."): "..tostring(msg):sub(1,300),true) end
                return
            end
            local ok2, dec = pcall(function() return HttpService:JSONDecode(res.Body) end)
            if not ok2 or not dec then addMessage("system","Resposta inválida (JSON).",true) return end
            local content=nil
            if dec.choices and dec.choices[1] and dec.choices[1].message and dec.choices[1].message.content then
                content=dec.choices[1].message.content
            elseif dec.content and dec.content[1] and dec.content[1].text then
                content=dec.content[1].text
            end
            if not content then addMessage("system","Resposta sem content.",true) return end
            addMessage("assistant",content)
            table.insert(State.chatHistory,{role="assistant",content=content})
        end)
    end
    sendBtn.MouseButton1Click:Connect(onSend)
    inputBox.FocusLost:Connect(function(enter) if enter then onSend() end end)

    local cName=tbox(custFrame,"Nome",UDim2.new(1,0,0,24),false)
    local cId=tbox(custFrame,"Model ID",UDim2.new(1,0,0,24),false)
    local cProv=tbox(custFrame,"Provedor",UDim2.new(1,0,0,24),false)
    local cPrice=btn(custFrame,"Preço: Desconhecido",UDim2.new(1,0,0,24),BG2) cPrice.TextSize=11
    local cUrl=tbox(custFrame,"API URL (opcional)",UDim2.new(1,0,0,24),false)
    local cSave=btn(custFrame,"Salvar modelo personalizado",UDim2.new(1,0,0,26),State.visual.accent)
    local priceOpts={"Gratuito","Pago","Desconhecido"} local priceIdx=3
    cPrice.MouseButton1Click:Connect(function() priceIdx=priceIdx%#priceOpts+1 cPrice.Text="Preço: "..priceOpts[priceIdx] end)

    local btnRow=Instance.new("Frame") btnRow.Size=UDim2.new(1,0,0,28) btnRow.BackgroundTransparency=1 btnRow.Parent=form
    local bL=Instance.new("UIListLayout") bL.FillDirection=Enum.FillDirection.Horizontal bL.Padding=UDim.new(0,6) bL.Parent=btnRow
    local saveBtn=btn(btnRow,"Salvar",UDim2.new(0,80,0,26),State.visual.accent)
    local cancelBtn=btn(btnRow,"Cancelar",UDim2.new(0,80,0,26),BG2)

    local editingId=nil

    local function getAllModels()
        local all={}
        for prov,list in pairs(Config.ModelsCatalog) do for _,m in ipairs(list) do table.insert(all,{id=m.id,name=m.name or m.id,provider=prov,price=m.price or "Unknown",isCustom=false}) end end
        for _,cm in ipairs(State.customModels) do table.insert(all,{id=cm.id,name=cm.name,provider=cm.provider or "Outros",price=cm.price or "Unknown",isCustom=true}) end
        return all
    end
    local function isFav(id) return State.favorites[id]==true end
    local function toggleFav(id) if State.favorites[id] then State.favorites[id]=nil else State.favorites[id]=true end Storage:save() end

    local function refreshModelList()
        for _,c in ipairs(modelList:GetChildren()) do if c:IsA("Frame") or c:IsA("TextButton") or c:IsA("TextLabel") then if c.Name~="UIListLayout" then c:Destroy() end end end
        local search=string.lower(modelSearch.Text or "")
        local all=getAllModels()
        local filtered={}
        for _,m in ipairs(all) do if search=="" or string.lower(m.id):find(search,1,true) or string.lower(m.name):find(search,1,true) or string.lower(m.provider):find(search,1,true) then table.insert(filtered,m) end end
        local groups={
            {name="[FAVORITOS]", key="fav", models={}},
            {name="[PROVEDOR: "..(selectedProvider or "TODOS").."]", key="prov", models={}},
            {name="[GRATUITOS]", key="free", models={}},
            {name="[PAGOS]", key="paid", models={}},
            {name="[OUTROS]", key="other", models={}},
            {name="[PERSONALIZADOS]", key="custom", models={}},
        }
        for _,m in ipairs(filtered) do
            if isFav(m.id) then table.insert(groups[1].models,m)
            elseif selectedProvider and m.provider==selectedProvider then table.insert(groups[2].models,m)
            elseif m.isCustom then table.insert(groups[6].models,m)
            elseif m.price=="Free" or m.price=="Gratuito" then table.insert(groups[3].models,m)
            elseif m.price=="Paid" or m.price=="Pago" then table.insert(groups[4].models,m)
            else table.insert(groups[5].models,m) end
        end
        local maxShow=120 local shown=0
        for _,g in ipairs(groups) do
            if #g.models>0 and shown<maxShow then
                if not (g.key=="prov" and not selectedProvider) then
                    local h=lbl(modelList,g.name.." ("..#g.models..")",UDim2.new(1,0,0,20),11,State.visual.accent) h.Font=Enum.Font.GothamBold shown=shown+1
                    table.sort(g.models,function(a,b) return a.name:lower()<b.name:lower() end)
                    for _,m in ipairs(g.models) do
                        if shown>=maxShow then break end
                        local row=Instance.new("Frame") row.Size=UDim2.new(1,0,0,26) row.BackgroundColor3=BG3 row.Parent=modelList corner(row,4)
                        lbl(row,m.name.." ("..m.id..") ["..m.price.."]",UDim2.new(1,-50,1,0),11,TXT).Position=UDim2.new(0,6,0,0)
                        local fav=btn(row,isFav(m.id) and "*" or "+",UDim2.new(0,22,0,22),BG2) fav.Position=UDim2.new(1,-46,0,2) fav.TextSize=12 fav.MouseButton1Click:Connect(function() toggleFav(m.id) refreshModelList() end)
                        local sel=btn(row,">",UDim2.new(0,22,0,22),State.visual.accent) sel.Position=UDim2.new(1,-22,0,2) sel.MouseButton1Click:Connect(function() selectedModel=m.id modelBtn.Text=m.id modelList.Visible=false end)
                        shown=shown+1
                    end
                end
            end
        end
        if shown==0 then lbl(modelList,"Nenhum modelo. Use pesquisa ou personalizado.",UDim2.new(1,0,0,24),11,TXTDIM) end
    end

    provBtn.MouseButton1Click:Connect(function() provList.Visible=not provList.Visible end)
    for _,prov in ipairs(Config.Providers) do
        local b=btn(provList,prov,UDim2.new(1,-4,0,24),BG3) b.TextSize=12
        b.MouseButton1Click:Connect(function() selectedProvider=prov provBtn.Text=prov provList.Visible=false if urlBox.Text=="" then local sug=Config.ProviderAPIUrls[prov] if sug and sug~="" then urlBox.Text=sug end end refreshModelList() end)
    end
    modelBtn.MouseButton1Click:Connect(function() refreshModelList() modelList.Visible=not modelList.Visible end)
    modelSearch:GetPropertyChangedSignal("Text"):Connect(function() if modelList.Visible then refreshModelList() end end)
    custBtn.MouseButton1Click:Connect(function() custFrame.Visible=not custFrame.Visible end)
    cSave.MouseButton1Click:Connect(function()
        if cId.Text=="" or cName.Text=="" then return end
        local map={Gratuito="Free", Pago="Paid", Desconhecido="Unknown"}
        local price=map[priceOpts[priceIdx]] or "Unknown"
        table.insert(State.customModels,{id=cId.Text,name=cName.Text,provider=cProv.Text~="" and cProv.Text or "Outros",price=price,apiUrl=cUrl.Text})
        Storage:save() custFrame.Visible=false cName.Text="" cId.Text="" cProv.Text="" cUrl.Text="" refreshModelList()
    end)

    updBtn.MouseButton1Click:Connect(function()
        updBtn.Text="..." task.spawn(function()
            if not selectedProvider then updBtn.Text="Selecione provedor" task.wait(1.5) updBtn.Text="Atualizar modelos" return end
            if selectedProvider=="OpenRouter" then
                local prof=getActiveProfile() local key=prof and prof.Provider=="OpenRouter" and prof.API_KEY or nil
                local headers={} if key then headers["Authorization"]="Bearer "..key end
                local res=httpRequest({Url="https://openrouter.ai/api/v1/models", Method="GET", Headers=headers})
                if res.Success then
                    local ok, dec=pcall(function() return HttpService:JSONDecode(res.Body) end)
                    if ok and dec and dec.data then
                        local cnt=0
                        for _,m in ipairs(dec.data) do
                            local exists=false for _,e in ipairs(State.customModels) do if e.id==m.id then exists=true break end end
                            if not exists then
                                local prov="Outros" if m.id:find("openai") then prov="OpenAI" elseif m.id:find("gemini") then prov="Google Gemini" elseif m.id:find("deepseek") then prov="DeepSeek" elseif m.id:find("claude") then prov="Anthropic" elseif m.id:find("qwen") then prov="Qwen" elseif m.id:find("mistral") then prov="Mistral" elseif m.id:find("llama") then prov="Meta / Llama" elseif m.id:find("grok") then prov="xAI / Grok" end
                                local price="Unknown" if m.pricing and m.pricing.prompt=="0" then price="Free" else price="Paid" end
                                table.insert(State.customModels,{id=m.id,name=m.name or m.id,provider=prov,price=price}) cnt=cnt+1
                            end
                        end
                        Storage:save() updBtn.Text="Atualizados "..cnt refreshModelList() task.wait(2) updBtn.Text="Atualizar modelos" return
                    end
                end
                updBtn.Text="Falha" task.wait(1.5) updBtn.Text="Atualizar modelos" return
            end
            -- OpenAI compat /models
            local base=urlBox.Text~="" and urlBox.Text or Config.ProviderAPIUrls[selectedProvider] or ""
            local mUrl=nil
            if base:find("/chat/completions") then mUrl=base:gsub("/chat/completions","/models")
            else local origin=base:match("^(https?://[^/]+)") if origin then mUrl=origin.."/v1/models" end end
            if not mUrl then updBtn.Text="Sem endpoint" task.wait(1.5) updBtn.Text="Atualizar modelos" return end
            local key=keyBox.Text~="" and keyBox.Text or (getActiveProfile() and getActiveProfile().API_KEY) or ""
            local headers={} if key~="" then headers["Authorization"]="Bearer "..key end
            local res=httpRequest({Url=mUrl, Method="GET", Headers=headers})
            if res.Success then
                local ok, dec=pcall(function() return HttpService:JSONDecode(res.Body) end)
                if ok and dec then
                    local list=dec.data or dec.models or {}
                    local cnt=0
                    for _,m in ipairs(list) do
                        local id=m.id or m.name
                        if id then
                            local exists=false for _,cat in pairs(Config.ModelsCatalog) do for _,cm in ipairs(cat) do if cm.id==id then exists=true break end end if exists then break end end if not exists then for _,cm in ipairs(State.customModels) do if cm.id==id then exists=true break end end end
                            if not exists then table.insert(State.customModels,{id=id,name=m.name or id,provider=selectedProvider,price="Unknown"}) cnt=cnt+1 end
                        end
                    end
                    Storage:save() updBtn.Text="Atualizados "..cnt refreshModelList() task.wait(2) updBtn.Text="Atualizar modelos" return
                end
            end
            updBtn.Text="Falha" task.wait(1.5) updBtn.Text="Atualizar modelos"
        end)
    end)

    local function refreshProfiles()
        for _,c in ipairs(listF:GetChildren()) do if c:IsA("Frame") then c:Destroy() end end
        for _,p in ipairs(State.profiles) do
            local row=Instance.new("Frame") row.Size=UDim2.new(1,0,0,36) row.BackgroundColor3=BG3 row.Parent=listF corner(row,6)
            lbl(row,p.Name.." | "..(p.Provider or "?").." | "..(p.MODEL or "sem modelo"),UDim2.new(1,-130,1,0),11,TXT).Position=UDim2.new(0,6,0,0)
            local use=btn(row,"Usar",UDim2.new(0,40,0,22),State.visual.accent) use.Position=UDim2.new(1,-120,0,7) use.TextSize=11
            use.MouseButton1Click:Connect(function() State.activeProfileId=p.id Storage:save() updateActiveDisplay() use.Text="Ativo" task.wait(1) use.Text="Usar" end)
            local edit=btn(row,"E",UDim2.new(0,22,0,22),BG2) edit.Position=UDim2.new(1,-72,0,7) edit.TextSize=11
            edit.MouseButton1Click:Connect(function() editingId=p.id nameBox.Text=p.Name urlBox.Text=p.API_URL keyBox.Text=p.API_KEY selectedProvider=p.Provider provBtn.Text=p.Provider or "Selecione provedor" selectedModel=p.MODEL modelBtn.Text=p.MODEL or "Selecione um modelo" form.Visible=true end)
            local del=btn(row,"X",UDim2.new(0,22,0,22),Color3.fromRGB(120,30,30)) del.Position=UDim2.new(1,-30,0,7) del.TextSize=11
            del.MouseButton1Click:Connect(function() for i,pr in ipairs(State.profiles) do if pr.id==p.id then table.remove(State.profiles,i) break end end if State.activeProfileId==p.id then State.activeProfileId=nil end Storage:save() refreshProfiles() refreshProfileDropdown() updateActiveDisplay() end)
        end
        if #State.profiles==0 then lbl(listF,"Nenhum perfil criado.",UDim2.new(1,0,0,24),11,TXTDIM) end
    end

    createBtn.MouseButton1Click:Connect(function() editingId=nil nameBox.Text="" urlBox.Text="" keyBox.Text="" selectedProvider=nil provBtn.Text="Selecione provedor" selectedModel=nil modelBtn.Text="Selecione um modelo" modelSearch.Text="" form.Visible=not form.Visible end)
    cancelBtn.MouseButton1Click:Connect(function() form.Visible=false editingId=nil end)
    saveBtn.MouseButton1Click:Connect(function()
        if nameBox.Text:gsub("%s","")=="" then return end
        local model=modelBtn.Text if model=="Selecione um modelo" then model="" end
        if selectedModel then model=selectedModel end
        if editingId then
            for _,p in ipairs(State.profiles) do if p.id==editingId then p.Name=nameBox.Text p.API_URL=urlBox.Text p.API_KEY=keyBox.Text p.Provider=selectedProvider or "Outros" p.MODEL=model break end end
        else
            local id=HttpService:GenerateGUID(false)
            table.insert(State.profiles,{id=id, Name=nameBox.Text, API_URL=urlBox.Text, API_KEY=keyBox.Text, Provider=selectedProvider or "Outros", MODEL=model})
            State.activeProfileId=id
        end
        Storage:save() form.Visible=false editingId=nil refreshProfiles() refreshProfileDropdown() updateActiveDisplay()
    end)

    refreshProfiles()
    refreshProfileDropdown()
    updateActiveDisplay()
    if #State.profiles==0 then addMessage("system","Configure a IA que você vai usar nas configurações.",true) end

    setTab("Chat")
    return sg
end

-- Init
local function init()
    Storage:load()
    local ok, guiOrErr = pcall(createGui)
    if not ok then
        warn("[AIChat] Erro ao criar GUI: "..tostring(guiOrErr))
        -- tenta mostrar erro simples no PlayerGui
        pcall(function()
            local pg = LocalPlayer and LocalPlayer:FindFirstChildOfClass("PlayerGui")
            if pg then
                local sg = Instance.new("ScreenGui") sg.Name="AIChat_Error" sg.Parent=pg
                local f = Instance.new("Frame") f.Size=UDim2.new(0,300,0,100) f.Position=UDim2.new(0.5,-150,0.5,-50) f.BackgroundColor3=Color3.fromRGB(30,30,30) f.Parent=sg
                local l = Instance.new("TextLabel") l.Size=UDim2.new(1,-20,1,-20) l.Position=UDim2.new(0,10,0,10) l.BackgroundTransparency=1 l.Text="AIChat erro: "..tostring(guiOrErr) l.TextColor3=Color3.new(1,1,1) l.TextWrapped=true l.Parent=f
            end
        end)
    else
        print("[AIChat] Inicializado OK - Perfis:", #State.profiles)
    end
end

task.spawn(function()
    local ok, err = pcall(init)
    if not ok then warn("[AIChat] Falha init: "..tostring(err)) end
end)
